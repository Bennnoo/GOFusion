cmake_minimum_required(VERSION 3.24...3.30)
project(okvis VERSION 2.0.0 LANGUAGES C CXX CUDA)
if(POLICY CMP0146)
  cmake_policy(SET CMP0146 OLD)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(OKVIS_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(DO_TIMING "Record runtime timing" ON)
set(BUILD_APPS ON)
set(SE_TEST OFF CACHE BOOL "Build Supereight tests")
set(SE_APP OFF CACHE BOOL "Build Supereight application")
set(OKVIS_ENABLE_GS_MAPPING ON)
set(OKVIS_PUBLIC_CXX_FLAGS)
set(OKVIS_PRIVATE_CXX_FLAGS -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake" "/usr/share/cmake/geographiclib")
set(BLA_VENDOR Generic)
find_package(LAPACK REQUIRED)
find_package(OpenCV 4.8 REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Torch REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem)
find_package(PCL REQUIRED)
find_package(GeographicLib REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} ${TORCH_CXX_FLAGS}")
add_compile_definitions(OKVIS_ENABLE_GS_MAPPING=1 OKVIS_USE_CUDA)
if(DO_TIMING)
  add_compile_definitions(OKVIS_TIMING)
endif()
add_subdirectory(external)
add_subdirectory(supereight-2-srl)
foreach(module util kinematics time cv common timing ceres frontend mapping multisensor_processing)
  add_subdirectory(okvis_${module})
endforeach()
add_executable(okvis2x_app_synchronous
  okvis_apps/src/okvis2x_app_synchronous.cpp
  okvis_apps/src/OkvisOccupancyViewer3D.cpp)
target_include_directories(okvis2x_app_synchronous PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(okvis2x_app_synchronous PRIVATE
  okvis::MultisensorProcessing ${TORCH_LIBRARIES} ${GeographicLib_LIBRARIES}
  ${PCL_LIBRARIES} Threads::Threads)
configure_file(resources/small_voc.yml.gz small_voc.yml.gz COPYONLY)
